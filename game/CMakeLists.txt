cmake_minimum_required(VERSION 3.16)
project(NeonCoast CXX C)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5)
FetchContent_MakeAvailable(raylib)
add_executable(NeonCoast WIN32 src/main.cpp)
target_link_libraries(NeonCoast raylib)
if(MSVC)
  set_target_properties(NeonCoast PROPERTIES LINK_FLAGS "/ENTRY:mainCRTStartup")
endif()
